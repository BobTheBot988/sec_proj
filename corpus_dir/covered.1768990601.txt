  7 |     |     /// @param interfaceID The interface identifier, as specified in ERC-165

  9 |     |     ///  uses less than 30,000 gas.
 10 |     |     /// @return `true` if the contract implements `interfaceID` and

 15 |     | contract ERC165Query {
 16 |     |     bytes4 constant InvalidID = 0xffffffff;
 17 |     |     bytes4 constant ERC165ID = 0x01ffc9a7;
 18 |     | 
 19 |     |     function doesContractImplementInterface(address _contract, bytes4 _interfaceId) external view returns (bool) {
 20 |     |         uint256 success;
 21 |     |         uint256 result;
 22 |     | 
 23 |     |         (success, result) = noThrowCall(_contract, ERC165ID);
 24 |     |         if ((success == 0) || (result == 0)) {
 25 |     |             return false;
 26 |     |         }
 27 |     | 
 28 |     |         (success, result) = noThrowCall(_contract, InvalidID);
 29 |     |         if ((success == 0) || (result != 0)) {
 30 |     |             return false;
 31 |     |         }
 32 |     | 
 33 |     |         (success, result) = noThrowCall(_contract, _interfaceId);
 34 |     |         if ((success == 1) && (result == 1)) {
 35 |     |             return true;
 36 |     |         }
 37 |     |         return false;
 38 |     |     }
 39 |     | 
 40 |     |     function noThrowCall(address _contract, bytes4 _interfaceId)
 41 |     |         internal
 42 |     |         view
 43 |     |         returns (uint256 success, uint256 result)
 44 |     |     {
 45 |     |         bytes4 erc165ID = ERC165ID;
 46 |     | 
 47 |     |         assembly {
 48 |     |             let x := mload(0x40) // Find empty storage location using "free memory pointer"
 49 |     |             mstore(x, erc165ID) // Place signature at beginning of empty storage
 50 |     |             mstore(add(x, 0x04), _interfaceId) // Place first argument directly next to signature
 51 |     | 
 52 |     |             success := staticcall(
 53 |     |                 30000, // 30k gas
 54 |     |                 _contract, // To addr
 55 |     |                 x, // Inputs are stored at location x
 56 |     |                 0x24, // Inputs are 36 bytes long
 57 |     |                 x, // Store output over input (saves space)
 58 |     |                 0x20
 59 |     |             ) // Outputs are 32 bytes long
 60 |     | 
 61 |     |             result := mload(x) // Load the result
 62 |     |         }
 63 |     |     }
